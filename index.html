async function toggleCamera() { const video = document.getElementById('webcamPreview'); const btn = event.target; if (!cameraActive) { try { mediaStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user' }, audio: true }); video.srcObject = mediaStream; cameraActive = true; btn.textContent = "🛑 Stop Live Stream Feed"; btn.style.background = "#ef4444"; btn.style.color = "#fff"; } catch (err) { // Log exact error to terminal instead of generic alert switchTab('terminal', document.querySelectorAll('.nav-item')[1]); const logs = document.getElementById('terminalLogs'); logs.innerHTML += `
> [ERROR]: Media stream failed - ${err.name}: ${err.message}
`; logs.scrollTop = logs.scrollHeight; } } else { if (mediaStream) { mediaStream.getTracks().forEach(track => track.stop()); } video.srcObject = null; cameraActive = false; btn.textContent = "📹 Initialize Live Stream Feed"; btn.style.background = "var(--accent)"; btn.style.color = "#000"; } }
